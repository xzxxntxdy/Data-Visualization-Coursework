# 模型训练配置示例

# 这是一个可选的配置文件示例
# 你可以将这些值复制到命令行中使用

# ============================================
# 快速配置（推荐新手）
# ============================================

# 简化 CNN 模型，快速训练
# python train.py --model_type simple --num_epochs 20 --batch_size 32

# Transformer 模型，更好的精度
# python train.py --model_type transformer --num_epochs 30 --batch_size 16


# ============================================
# 数据相关
# ============================================

# --dataset_dir ./dataset             数据集目录
# --image_size 256                    输入图像大小（像素）
# --dummy_num_images 500              虚拟数据集图像数量


# ============================================
# 模型相关
# ============================================

# --model_type simple                 模型类型：simple 或 transformer
# --num_keypoints 17                  关键点数量（COCO 标准）
# --hidden_dim 768                    (Transformer) 隐层维度
# --num_heads 8                       (Transformer) 注意力头数
# --num_layers 12                     (Transformer) 编码器层数
# --patch_size 16                     (Transformer) Patch 大小


# ============================================
# 训练超参数
# ============================================

# --batch_size 32                     批大小
# --num_epochs 50                     总训练轮数
# --learning_rate 1e-4                初始学习率
# --weight_decay 1e-5                 L2 正则化系数
# --num_workers 4                     数据加载进程数


# ============================================
# 学习率调度
# ============================================

# --scheduler_step 10                 学习率调度步长（epoch）
# --scheduler_gamma 0.1               学习率衰减因子


# ============================================
# 保存和日志
# ============================================

# --checkpoint_dir ./models           模型保存目录
# --log_dir ./logs                    日志目录
# --save_interval 10                  检查点保存间隔（epoch）


# ============================================
# 完整命令示例
# ============================================

# 示例 1：快速测试（简化模型）
# python train.py \
#     --model_type simple \
#     --dataset_dir ./dataset \
#     --dummy_num_images 500 \
#     --num_epochs 20 \
#     --batch_size 32 \
#     --learning_rate 1e-4

# 示例 2：高精度模型（Transformer）
# python train.py \
#     --model_type transformer \
#     --dataset_dir ./dataset \
#     --dummy_num_images 1000 \
#     --num_epochs 50 \
#     --batch_size 16 \
#     --hidden_dim 768 \
#     --num_heads 8 \
#     --num_layers 12 \
#     --learning_rate 1e-4

# 示例 3：使用自己的数据
# python train.py \
#     --model_type simple \
#     --dataset_dir ~/my_dataset \
#     --num_epochs 30 \
#     --batch_size 32
